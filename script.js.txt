// Fade-in on scroll
document.addEventListener("scroll", function () {
  const fadeElems = document.querySelectorAll(".fade-in");
  fadeElems.forEach(el => {
    const rect = el.getBoundingClientRect();
    if (rect.top < window.innerHeight - 100) el.classList.add("visible");
  });
});

// Scroll particles
const particlesContainer = document.querySelector(".scroll-particles");
document.addEventListener("scroll", function() {
  const particle = document.createElement("div");
  particle.classList.add("particle");
  const x = (Math.random() - 0.5) * 100; // random horizontal movement
  const y = Math.random() * 100 + 50;   // random vertical movement
  particle.style.setProperty("--x", `${x}px`);
  particle.style.setProperty("--y", `${y}px`);
  particle.style.left = Math.random() * window.innerWidth + "px";
  particle.style.top = window.scrollY + "px";
  particlesContainer.appendChild(particle);
  setTimeout(() => particle.remove(), 3000);
});
